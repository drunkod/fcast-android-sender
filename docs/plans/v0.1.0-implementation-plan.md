### Step 3 — SRT arm in `build_live_pipeline`

**File:** `crates/migration-runtime/src/nodes/destination.rs`
**Inside** `DestinationNode::build_live_pipeline`, after the
`DestinationFamily::Udp` arm (around line 865).

The SRT pipeline is structurally identical to UDP — `mpegtsmux → srtsink`
instead of `mpegtsmux → udpsink` — with additional SRT properties.

```rust
DestinationFamily::Srt {
    uri,
    latency,
    passphrase,
    pbkeylen,
} => {
    let mux  = Self::make_element("mpegtsmux", None)?;
    let sink = Self::make_element("srtsink",   None)?;

    pipeline.add(&mux).map_err(|err| {
        format!("Failed to add mpegtsmux to srt pipeline: {err:?}")
    })?;
    pipeline.add(&sink).map_err(|err| {
        format!("Failed to add srtsink to srt pipeline: {err:?}")
    })?;

    // ── mpegtsmux: 188-byte MPEG-TS packet alignment ────────────────
    // Without alignment=7 some receivers (FFmpeg) report continuity errors.
    if mux.has_property("alignment") {
        mux.set_property("alignment", 7i32);
    }

    // ── srtsink properties ──────────────────────────────────────────
    sink.set_property("uri", uri.clone());
    sink.set_property("latency", *latency);  // i32 milliseconds

    if let (Some(pass), Some(keylen)) = (passphrase.as_deref(), pbkeylen) {
        sink.set_property("passphrase", pass);
        sink.set_property("pbkeylen", *keylen);
    }

    // ── Video chain ─────────────────────────────────────────────────
    if let Some(appsrc) = video_appsrc.as_ref() {
        let vconv      = Self::make_element("videoconvert", None)?;
        let venc_chain = Self::select_video_encoder(&self.id)?;
        let vparse     = Self::make_element("h264parse",    None)?;

        pipeline.add(&vconv).map_err(|err| {
            format!("Failed to add videoconvert to srt pipeline: {err:?}")
        })?;
        Self::add_video_encoder_chain(&pipeline, &venc_chain, "srt pipeline")?;
        pipeline.add(&vparse).map_err(|err| {
            format!("Failed to add h264parse to srt pipeline: {err:?}")
        })?;

        gst::Element::link_many(
            [appsrc.upcast_ref::<gst::Element>(), &vconv].as_slice(),
        )
        .map_err(|err| format!("Failed to link srt video pre-processing: {err:?}"))?;

        Self::link_video_encoder_chain(
            &vconv,
            &venc_chain,
            &vparse,
            "srt video encoder chain",
        )?;

        gst::Element::link_many([&vparse, &mux].as_slice())
            .map_err(|err| format!("Failed to link srt video output: {err:?}"))?;
    }

    // ── Audio chain ─────────────────────────────────────────────────
    if let Some(appsrc) = audio_appsrc.as_ref() {
        let aconv    = Self::make_element("audioconvert",  None)?;
        let aresample = Self::make_element("audioresample", None)?;
        let aenc      = Self::make_element("avenc_aac",     None)?;

        pipeline.add(&aconv).map_err(|err| {
            format!("Failed to add audioconvert to srt pipeline: {err:?}")
        })?;
        pipeline.add(&aresample).map_err(|err| {
            format!("Failed to add audioresample to srt pipeline: {err:?}")
        })?;
        pipeline.add(&aenc).map_err(|err| {
            format!("Failed to add avenc_aac to srt pipeline: {err:?}")
        })?;

        gst::Element::link_many(
            [
                appsrc.upcast_ref::<gst::Element>(),
                &aconv,
                &aresample,
                &aenc,
                &mux,
            ]
            .as_slice(),
        )
        .map_err(|err| format!("Failed to link srt audio chain: {err:?}"))?;
    }

    mux.link(&sink)
        .map_err(|err| format!("Failed to link mpegtsmux to srtsink: {err:?}"))?;
}
```

**Pipeline diagram:**

```
appsrc(video) → videoconvert → [capsfilter] → amcvidenc/x264enc → h264parse ─┐
                                                                               ▼
                                                                          mpegtsmux → srtsink
                                                                               ▲
appsrc(audio) → audioconvert → audioresample → avenc_aac ─────────────────────┘
```

```rust
#[test]
fn srt_destination_defaults_latency_when_omitted() {
    let cmd: Command = serde_json::from_str(
        r#"{"createdestination":{"id":"srt-1","family":{"Srt":{"uri":"srt://10.0.0.1:9000"}}}}"#,
    )
    .unwrap();
    match cmd {
        Command::CreateDestination {
            family: DestinationFamily::Srt { latency, passphrase, pbkeylen, .. },
            ..
        } => {
            assert_eq!(latency, 200);
            assert!(passphrase.is_none());
            assert!(pbkeylen.is_none());
        }
        other => panic!("expected Srt destination, got {other:?}"),
    }
}

#[test]
fn srt_destination_with_encryption_roundtrip() {
    let family = DestinationFamily::Srt {
        uri: "srt://10.0.0.1:9000".to_string(),
        latency: 500,
        passphrase: Some("supersecretphrase".to_string()),
        pbkeylen: Some(32),
    };
    let json    = serde_json::to_string(&family).unwrap();
    let decoded: DestinationFamily = serde_json::from_str(&json).unwrap();
    assert_eq!(decoded, family);

#[test]
fn srt_destination_passphrase_absent_omitted_from_wire() {
    let family = DestinationFamily::Srt {
        uri: "srt://10.0.0.1:9000".to_string(),
        latency: 200,
        passphrase: None,
        pbkeylen: None,
    };
    let json = serde_json::to_string(&family).unwrap();

fn destination_family_roundtrip_includes_srt() {
    let family = DestinationFamily::Srt {

        pbkeylen: None,
    };
    let encoded = serde_json::to_string(&family).unwrap();
    let decoded: DestinationFamily = serde_json::from_str(&encoded).unwrap();

}
```

```rust
#[test]
fn srt_profile_lists_srtsink_and_mpegtsmux() {
```

### Step 6 — RTMP settings page (`ui/pages/protocol_rtmp_settings_page.slint`)

**Moblin analogue:** `View/Settings/Streams/Stream/Rtmp/StreamRtmpSettingsView.swift`

Moblin's RTMP settings view is intentionally minimal — a single adaptive
bitrate toggle. FCast adds the RTMP URL field (already in
`cam-rtmp-url` / `cam-rtmp-stream-key` on the Bridge, surfaced here as
a read-only status row that links back to `camera-rtmp-stream`).

```slint
// ui/pages/protocol_rtmp_settings_page.slint
import { Theme } from "../theme.slint";
import { Bridge, Panel } from "../bridge.slint";
import { PanelBridge } from "../state/panel_bridge.slint";
import { PanelHeader } from "../components/panel_chrome.slint";
import {
    SettingsSection,
    SettingsValueRow,
    SettingsToggleRow,
    SettingsBadgeRow,
} from "../components/settings_rows.slint";

export component ProtocolRtmpSettingsPage inherits Rectangle {
    width: 100%;
    height: 100%;
    background: transparent;

    // Local state
    in-out property <bool>   adaptive-bitrate: false;
    in-out property <bool>   reconnect-on-disconnect: true;
    in-out property <bool>   show-stream-key: false;

    forward-focus: scope;
    scope := FocusScope {
        key-pressed(event) => {
            if (event.text == Key.Escape) { PanelBridge.pop(); return accept; }
            return reject;
        }

        VerticalLayout {
            PanelHeader {
                title: @tr("RTMP");
                close-clicked => { PanelBridge.pop(); }
            }

            Flickable {
                vertical-stretch: 1;
                viewport-height: content.preferred-height;

                content := VerticalLayout {
                    padding: Theme.padding-screen;
                    spacing: Theme.spacing-loose;
                    alignment: start;

                    // ── Connection ──────────────────────────────────────────
                    SettingsSection { title: @tr("Connection"); }
                    SettingsSection {
                        SettingsValueRow {
                            title: @tr("Server URL");
                            value: Bridge.cam-rtmp-url != "" ? Bridge.cam-rtmp-url : @tr("Not set");
                            clicked => { PanelBridge.push(Panel.camera-rtmp-stream); }
                        }
                        SettingsValueRow {
                            title: @tr("Stream Key");
                            value: root.show-stream-key
                                ? (Bridge.cam-rtmp-stream-key != ""
                                      ? Bridge.cam-rtmp-stream-key
                                      : @tr("Not set"))
                                : "●●●●●●●●";
                            clicked => { root.show-stream-key = !root.show-stream-key; }
                        }
                    }

                    // ── Reliability ─────────────────────────────────────────
                    SettingsSection { title: @tr("Reliability"); }
                    SettingsSection {
                        SettingsToggleRow {
                            title: @tr("Reconnect on disconnect");
                            value: root.reconnect-on-disconnect;
                            toggled(v) => { root.reconnect-on-disconnect = v; }
                        }
                    }

                    // ── Performance ─────────────────────────────────────────
                    SettingsSection { title: @tr("Performance"); }
                    SettingsSection {
                        SettingsToggleRow {
                            title: @tr("Adaptive bitrate");
                            subtitle: @tr("Reduce bitrate when network is congested (v0.2.0)");
                            value: root.adaptive-bitrate;
                            enabled: false; // v0.2.0
                            toggled(v) => { root.adaptive-bitrate = v; }
                        }
                    }

                    // ── Status ──────────────────────────────────────────────
                    if Bridge.cam-rtmp-state != MixerState.idle: SettingsSection {
                        title: @tr("Stream Status");
                    }
                    if Bridge.cam-rtmp-state != MixerState.idle: SettingsSection {
                        SettingsBadgeRow {
                            title: @tr("State");
                            value: Bridge.cam-rtmp-state == MixerState.running
                                ? @tr("Live") : @tr("Connecting…");
                        }
                        if Bridge.cam-rtmp-error-text != "": SettingsValueRow {
                            title: @tr("Error");
                            value: Bridge.cam-rtmp-error-text;
                        }
                    }
                }
            }
        }
    }
}
```

## 3. Rust handler sketch (non-normative)

The Slint callbacks need Rust implementations in
`src/backend/migration_backend.rs`. This is a reference sketch for the
`save-srt-destination-config` and `start-srt-destination` flows:

```rust
// Inside the migration backend callback wiring (not a new file):

let bridge = ui.global::<Bridge>();

// save-srt-destination-config: persist URI + latency to app config
bridge.on_save_srt_destination_config({
    let ui_weak = ui.as_weak();
    move || {
        let ui = ui_weak.upgrade().unwrap();
        let b  = ui.global::<Bridge>();
        let dest = b.get_srt_destination();
        let passphrase = b.get_srt_destination_passphrase().to_string();
        // ... persist to config store
    }
});

// start-srt-destination: build the migration-runtime graph command
bridge.on_start_srt_destination({
    let ui_weak = ui.as_weak();
    let tx = runtime_tx.clone();
    move || {
        let ui   = ui_weak.upgrade().unwrap();
        let dest = ui.global::<Bridge>().get_srt_destination();
        let pass = ui.global::<Bridge>().get_srt_destination_passphrase();

        // Build the CreateDestination command JSON
        let family_json = if !pass.is_empty() {
            serde_json::json!({
                "Srt": {
                    "uri":       dest.uri.to_string(),
                    "latency":   dest.latency_ms,
                    "passphrase": pass.to_string(),
                    "pbkeylen":  16  // from encryption-idx
                }
            })
        } else {
            serde_json::json!({
                "Srt": {
                    "uri":     dest.uri.to_string(),
                    "latency": dest.latency_ms
                }
            })
        };

        let cmd = serde_json::json!({
            "createdestination": {
                "id":     "srt-dest-1",
                "family": family_json,
                "audio":  true,
                "video":  true
            }
        });

        let _ = tx.send(cmd.to_string());
    }
});
```

## 5. Pitfalls catalogue

These are adapted from `MVP-PHASE-8-srt-destination-family.md` and the
Moblin SRT implementation notes.

| # | Pitfall | Symptom | Fix |
|---|---|---|---|
| P1 | `srtsink` not found at runtime | `Could not create element of type srtsink` | Already in `Android.mk:63` — verify registry with `adb logcat \| grep srtsink` |
| P2 | Caller vs listener mismatch | Destination sits in Starting for 3s then stops | Append `?mode=listener` to URI when you want the phone to accept inbound connections |
| P3 | passphrase without pbkeylen | Silent connect then immediate close on peer side | Both must be set together — validate in Rust before building the pipeline |
| P4 | Latency mismatch between endpoints | Unexpected +N second end-to-end delay | SRT silently takes the larger of the two values — use the same value on both ends |
| P5 | `mpegtsmux alignment` not set | FFmpeg receiver reports `continuity counter` errors on every packet | Set `mux.set_property("alignment", 7i32)` (mirroring the UDP arm) |
| P6 | IPv6 URI without brackets | `Invalid URI: srt://fe80::1:1234` | Use `srt://[fe80::1]:1234` — brackets required |
| P7 | `latency` passed as `i64` | `cannot set property 'latency' of type 'gint' from gint64` | Property type is `i32` (milliseconds), not nanoseconds |
| P8 | Passphrase shorter than 10 chars | Handshake silently fails | SRT spec: 10–79 ASCII characters; validate before setting property |
| P9 | Non-exhaustive `match` after adding `Srt` variant | Compile error in `protocol.rs`, `destination.rs` | Steps 2 and 3 must land in the same commit as Step 1 |
